# Visuals Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** A Claude session calls blimp's new `show` tool with a web page it wrote; blimpd keeps a copy; the iPhone's
chat shows a card that opens the page full screen, and the Mac opens it in a tab in that project, both locked down
(no requests, no internet, an opaque origin) and styled by blimp's kit (its colours, Chart.js, Mermaid).

**Architecture:** A new blimpd module, `src/visuals/`: the copies and the visual key (`store.ts`), the kit built into
the binary (`kit.ts`, `kit/`), the page and kit responses with their headers (`http.ts`) and the stdio MCP server
(`mcp.ts`, `blimpd visuals-mcp`). `server.ts` adds `GET /visuals/key`, `POST /visuals` (which publishes a `visual`
event) and the key-in-path reads. Every session gets the MCP through `<socket dir>/mcp.json`. `chat.ts` turns a
show call into a `visual` chat message. The Mac adds a `visual:<source>` tab kind with a sandboxed frame; the iPhone
adds a card and a full-screen `WKWebView`.

**Tech Stack:** Bun 1.4 + TypeScript (blimpd, `bun test`); React 19 + Vite + Tauri 2 (Mac, `bun test src`,
Playwright e2e in Chromium); SwiftUI + WebKit, iOS 26, Swift 6 (app); BlimpKit (Swift package, Swift Testing,
Foundation only, also tested on Linux); XCUITest; GitHub Actions `.github/workflows/ios.yml`.

**Spec:** `docs/superpowers/specs/2026-10-26-visuals-design.md`

```ts
// Visuals (docs/superpowers/specs/2026-10-26-visuals-design.md): pages a session showed with blimp's show tool
// (visuals/mcp.ts). Each is a copy of the page as it was then, <socket dir>/visuals/<id>.html, with <id>.json saying
// what it is, kept 30 days, so a card shows what you were shown even after the file changes. The visual key, which
// reads only /visuals/<key>/... (server.ts), comes from a token.

import { createHmac, randomBytes } from "node:crypto";
import { existsSync, mkdirSync, readdirSync, readFileSync, rmSync, statSync, writeFileSync } from "node:fs";
import { isAbsolute, join } from "node:path";
import { guardFile, HttpError } from "../files";
import { blimpDir } from "../sessions";

export type Visual = { id: string; title: string; session: string; dir: string; source: string; at: number };

export const MAX_PAGE = 2 * 1024 * 1024;
export const MAX_TITLE = 80;
export const KEEP_MS = 30 * 24 * 3600 * 1000;

export const isVisualId = (id: string) => /^[0-9a-f]{12}$/.test(id);
export const visualsDir = () => join(blimpDir(), "visuals");

const utf8 = (b: Buffer) => {
  try {
    new TextDecoder("utf-8", { fatal: true }).decode(b);
    return true;
  } catch {
    return false;
  }
};

// The page at path (absolute, or relative to dir, the session's folder) for session, copied. Under home and never
// one of blimp's own files, as every file route's (guardFile).
export function makeVisual(session: string, dir: string, path: string, title: string, now = Date.now()): Visual {
  const name = title.replace(/[\u0000-\u001f\u007f]/g, " ").trim();
  if (!name) throw new HttpError(400, "a title is required");
  if (name.length > MAX_TITLE) throw new HttpError(400, `the title must be at most ${MAX_TITLE} characters`);
  const source = guardFile(isAbsolute(path) ? path : join(dir, path));
  let size: number;
  try {
    const st = statSync(source);
    if (!st.isFile()) throw new HttpError(400, `not a file: ${path}`);
    size = st.size;
  } catch (e) {
    if (e instanceof HttpError) throw e;
    throw new HttpError(404, `no such file: ${path}`);
  }
  if (size > MAX_PAGE) throw new HttpError(413, `the page is ${(size / 1048576).toFixed(1)} MB, over 2 MB: draw data with the kit, not big images`);
  const bytes = readFileSync(source);
  if (!utf8(bytes)) throw new HttpError(400, "the page isn't UTF-8 text");
  const v: Visual = { id: randomBytes(6).toString("hex"), title: name, session, dir, source, at: now };
  mkdirSync(visualsDir(), { recursive: true, mode: 0o700 });
  writeFileSync(join(visualsDir(), `${v.id}.html`), bytes, { mode: 0o600 });
  writeFileSync(join(visualsDir(), `${v.id}.json`), JSON.stringify(v) + "\n", { mode: 0o600 });
  return v;
}

export function visualPage(id: string): Buffer | undefined {
  if (!isVisualId(id)) return;
  try {
    return readFileSync(join(visualsDir(), `${id}.html`));
  } catch {}
}

export const isKept = (id: string) => isVisualId(id) && existsSync(join(visualsDir(), `${id}.html`));

// Copies written over 30 days ago go: when blimpd starts, and once a day (server.ts).
export function cleanVisuals(now = Date.now()) {
  let names: string[];
  try {
    names = readdirSync(visualsDir());
  } catch {
    return;
  }
  for (const n of names) {
    const p = join(visualsDir(), n);
    try {
      if (now - statSync(p).mtimeMs > KEEP_MS) rmSync(p, { force: true });
    } catch {}
  }
}

// HMAC-SHA256 of a token (the Mac's or the phone's) with the text "blimp visuals", hex. No token: none.
export const visualKey = (token: string) => (token ? createHmac("sha256", token).update("blimp visuals").digest("hex") : "");
```

```js
// scripts: it sets Chart.js's font, colours and tooltips from blimp.css, gives datasets with no colours of their own

(() => {
  const css = (name) => getComputedStyle(document.documentElement).getPropertyValue(name).trim();
  const series = () => ["--c1", "--c2", "--c3", "--c4", "--c5", "--c6"].map(css);
  const tint = (hex, a) => {
    const n = parseInt(hex.slice(1), 16);
    return `rgba(${n >> 16}, ${(n >> 8) & 255}, ${n & 255}, ${a})`;

  const colors = {
    id: "blimpColors",
    beforeUpdate(chart) {
      const c = series();
      const round = ["pie", "doughnut", "polarArea"].includes(chart.config.type);
      chart.data.datasets.forEach((d, i) => {
        if (!ours.has(d) && (d.backgroundColor || d.borderColor)) return;
        ours.add(d);
        if (round) {
          d.backgroundColor = d.data.map((_, j) => c[j % c.length]);
          d.borderColor = css("--bg");
        } else {
          const color = c[i % c.length];
          d.borderColor = color;
          d.backgroundColor = (d.type ?? chart.config.type) === "line" ? tint(color, 0.15) : color;
        }
      });
    },
  };
```

```ts
// blimp's kit for visuals (src/visuals/kit/), built into blimpd's binary (bun's file imports): its stylesheet and
// script, Chart.js and Mermaid (LICENSES.md). A page loads them by relative paths, kit/<file>, which carry the visual
// key (visuals/http.ts). Each is gzipped once, the first time it's asked for: Mermaid is 5.5 MB, 1.6 MB gzipped.

import blimpCss from "./kit/blimp.css" with { type: "file" };
import blimpJs from "./kit/blimp.js" with { type: "file" };
import chartJs from "./kit/chart.js" with { type: "file" };
import mermaidJs from "./kit/mermaid.js" with { type: "file" };

const JS = "text/javascript; charset=utf-8";
export const KIT: Record<string, { file: string; type: string }> = {
  "blimp.css": { file: blimpCss, type: "text/css; charset=utf-8" },
  "blimp.js": { file: blimpJs, type: JS },
  "chart.js": { file: chartJs, type: JS },
  "mermaid.js": { file: mermaidJs, type: JS },
};

// The libraries as vendored (test/visuals-kit.test.ts): Chart.js 4.5.1 and Mermaid 12.1.0.
export const KIT_SUMS = {
  "chart.js": "48444a82d4edcb5bec0f1965faacdde18d9c17db3063d042abada2f705c9f54a",
  "mermaid.js": "6484afc32872a3aa16cac9a76ba1816a1ed4cc870a6593cc2e17757750f518b2",
};

const read = new Map<string, Promise<{ bytes: Uint8Array; gzipped: Uint8Array; type: string }>>();

export function kitFile(name: string) {
  if (!Object.hasOwn(KIT, name)) return Promise.resolve(undefined);
  let f = read.get(name);
  if (!f) {
    const { file, type } = KIT[name];
    read.set(name, (f = Bun.file(file).bytes().then((bytes) => ({ bytes, gzipped: Bun.gzipSync(bytes), type }))));
  }
  return f;
}
```

```ts
// `blimpd visuals-mcp`: blimp's own MCP server, which every session blimp starts gets as "blimp" (browser/mcp.ts's
// mcp.json). One tool, show: Claude hands it a page it wrote, and blimpd keeps a copy and shows it on the phone and
// the Mac (POST /visuals, visuals/store.ts). MCP over stdio is JSON-RPC 2.0, one message per line; this answers the
// few methods a client calls, with no MCP library. It finds its session from ZMX_SESSION, as browser-mcp does, and
// reaches blimpd on its socket with the Mac's token, as `blimpd send` does.

import { readFileSync } from "node:fs";
import { join } from "node:path";
import pkg from "../../package.json";
import { blimpDir, socketPath } from "../sessions";

export const SHOW_DESCRIPTION = `Show the user a visual on their phone and Mac: a chart, a diagram, a table, a timeline or a small report, as one HTML page you wrote. Use it when a picture explains better than text, or when the user asks to see something.

Write the page under the home folder, not /tmp (blimp can't read there): in the project if it should be kept, else in ~/.cache/blimp-visuals/. Then call show with its path and a short title. After you change the file, call show again: each call shows a new copy.

The page has no internet and can make no requests. Load only blimp's kit, by these relative paths, in <head>:
  <link rel="stylesheet" href="kit/blimp.css">   blimp's colours for light and dark, the system font, and .grid, .card, .stats with .stat (<b>42</b><span>label</span>), .chart, .scroll, tables (td.num), .tag, .good/.warn/.bad

  <script src="kit/blimp.js"></script>          after those: sets them to blimp's look, draws the diagrams, follows light and dark

Start with <meta name="viewport" content="width=device-width, initial-scale=1">. Text at least 14 px. Nothing that works only on hover: a tap on a chart shows its values. Give each chart a box: <div class="chart"><canvas></canvas></div>, with the Chart.js option maintainAspectRatio: false. Use the kit's colours (var(--c1) to var(--c6) for series, --fg, --fg-dim, --line); datasets without colours get them. At most 2 MB.`;

const TOOLS = [
  {
    name: "show",
    description: SHOW_DESCRIPTION,
    inputSchema: {
      type: "object",
      properties: {
        path: { type: "string", description: "The page: an absolute path, or one relative to the session's folder. Under the home folder." },
        title: { type: "string", description: "A short title for the card and the tab, like 'Test runs this week'.", maxLength: 80 },
      },
      required: ["path", "title"],
      additionalProperties: false,
    },
  },
];

type Show = (path: string, title: string) => Promise<string>;

// The answer to one message, or undefined for a notification (no id), which gets none.
export async function handle(msg: unknown, show: Show): Promise<object | undefined> {
  const { id, method, params } = (msg ?? {}) as { id?: string | number | null; method?: string; params?: any };
  if (id === undefined || id === null) return;
  const result = (r: object) => ({ jsonrpc: "2.0", id, result: r });
  const failure = (code: number, message: string) => ({ jsonrpc: "2.0", id, error: { code, message } });
  switch (method) {
    case "initialize":
      // The client's own version back: this server uses nothing that differs between them.
      return result({
        protocolVersion: typeof params?.protocolVersion === "string" ? params.protocolVersion : "2025-06-18",
        capabilities: { tools: {} },
        serverInfo: { name: "blimp", version: pkg.version },
      });
    case "ping":
      return result({});
    case "tools/list":
      return result({ tools: TOOLS });
    case "tools/call": {
      if (params?.name !== "show") return failure(-32602, `no tool called ${params?.name}`);
      const { path, title } = params.arguments ?? {};
      try {
        const shown = await show(typeof path === "string" ? path : "", typeof title === "string" ? title : "");
        return result({ content: [{ type: "text", text: `Shown on the phone and the Mac. visual: ${shown}` }] });
      } catch (e) {
        return result({ content: [{ type: "text", text: `Not shown: ${(e as Error).message}` }], isError: true });
      }
    }
    default:
      return failure(-32601, `no method ${method}`);
  }
}

// POST /visuals on blimpd's socket, with the Mac's token: the new copy's id, or blimpd's reason as the error.
async function postShow(session: string, path: string, title: string): Promise<string> {
  let token: string;
  try {
    token = readFileSync(join(blimpDir(), "token"), "utf8").trim();
  } catch {
    throw new Error("blimpd isn't set up here (no token)");
  }
  let r: Response;
  try {
    r = await fetch("http://blimpd/visuals", {
      method: "POST",
      unix: socketPath(),
      headers: { Authorization: `Bearer ${token}`, "Content-Type": "application/json" },
      body: JSON.stringify({ session, path, title }),
    } as any);
  } catch {
    throw new Error("blimpd isn't running");
  }
  const body = (await r.json().catch(() => ({}))) as { id?: string; error?: string };
  if (!r.ok || !body.id) throw new Error(body.error ?? `blimpd answered ${r.status}`);
  return body.id;
}

export async function runVisualsMcp(): Promise<number> {
  const session = process.env.ZMX_SESSION;
  const show: Show = (path, title) =>
    session ? postShow(session, path, title) : Promise.reject(new Error("this runs inside a blimp session (ZMX_SESSION isn't set)"));
  // One line at a time, in order: an answer longer than a pipe's buffer must not be cut into by another.
  let out = Promise.resolve(0);
  const write = (m: object) => (out = out.then(() => Bun.write(Bun.stdout, JSON.stringify(m) + "\n")));
  const pending = new Set<Promise<unknown>>();
  const decoder = new TextDecoder();
  let buf = "";
  for await (const chunk of Bun.stdin.stream()) {
    buf += decoder.decode(chunk, { stream: true });
    for (let nl = buf.indexOf("\n"); nl >= 0; nl = buf.indexOf("\n")) {
      const line = buf.slice(0, nl).trim();
      buf = buf.slice(nl + 1);
      if (!line) continue;
      let msg: unknown;
      try {
        msg = JSON.parse(line);
      } catch {
        write({ jsonrpc: "2.0", id: null, error: { code: -32700, message: "not JSON" } });
        continue;
      }
      // A slow show doesn't hold up a ping behind it.
      const p = handle(msg, show).then((r) => r && write(r));
      pending.add(p);
      p.finally(() => pending.delete(p));
    }
  }
  await Promise.all(pending);
  await out;
  return 0;
}
```

```swift
// A page Claude showed with blimp's show tool (docs/superpowers/specs/2026-10-26-visuals-design.md): where its copy is
// read. The visual key (GET /visuals/key) is in the path, so the page's relative kit/ paths carry it.
import Foundation

public enum VisualLink {
    /// The page on `server` (the paired address, https://<tailnet name>), or nil for an id or key blimpd wouldn't make.
    public static func page(server: URL, key: String, id: String) -> URL? {
        guard id.range(of: "^[0-9a-f]{12}$", options: .regularExpression) != nil,
              key.range(of: "^[0-9a-f]{64}$", options: .regularExpression) != nil,
              var c = URLComponents(url: server, resolvingAgainstBaseURL: false) else { return nil }
        if c.path.hasSuffix("/") { c.path.removeLast() }
        c.path += "/visuals/\(key)/\(id)"
        c.query = nil
        c.fragment = nil
        return c.url
    }
}
```

```swift
/// blimpd's copy of a visual in a web view that goes nowhere else.
struct VisualPage: UIViewRepresentable {
    let url: URL

    @MainActor private static let data = WKWebsiteDataStore.nonPersistent()

    func makeCoordinator() -> Coordinator { Coordinator(url: url) }

    func makeUIView(context: Context) -> WKWebView {
        let config = WKWebViewConfiguration()
        config.websiteDataStore = Self.data
        let view = WKWebView(frame: .zero, configuration: config)
        view.navigationDelegate = context.coordinator
        view.allowsLinkPreview = false
        view.allowsBackForwardNavigationGestures = false
        view.isOpaque = false
        view.backgroundColor = .systemBackground
        view.accessibilityIdentifier = "visualPage"
        view.load(URLRequest(url: url))
        return view
    }

    func updateUIView(_ view: WKWebView, context: Context) {}

    @MainActor
    final class Coordinator: NSObject, WKNavigationDelegate {
        let url: URL
        init(url: URL) { self.url = url }

        /// The page's own address loads; a link, a script or a redirect sending it anywhere else is cancelled.
        func webView(_ webView: WKWebView, decidePolicyFor navigationAction: WKNavigationAction) async -> WKNavigationActionPolicy {
            navigationAction.request.url?.absoluteString == url.absoluteString ? .allow : .cancel
        }
    }
```
